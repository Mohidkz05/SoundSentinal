# Training dependencies for M3. Same pins as ../requirements.txt, built against
# CUDA instead of CPU — that file's `--extra-index-url .../cpu` would otherwise
# quietly give a compute node a CPU-only torch, and the job would run to
# completion at a tenth of the speed with nothing in the log to explain it.
#
#   uv pip install -r hpc/requirements-cuda.txt
#
# The `+cu130` local version is load-bearing, not decoration. An earlier version
# of this file asked for cu129 by index URL and got cu130 anyway: uv gives
# --extra-index-url priority over --index-url (the opposite of pip), so PyPI won
# the resolution and PyPI's torch 2.13.0 is itself a CUDA 13 build. Pinning the
# local version makes that impossible — PyPI's `2.13.0` carries no local tag and
# simply does not match `==2.13.0+cu130`, so only this index can satisfy it.
#
# Why CUDA 13 is the right target *on M3 specifically*: the GPU nodes run driver
# 580.126.20, and CUDA 13 wants >= 580. Verified on the hardware rather than
# inferred — a real matmul and a MelSpectrogram both ran on an L40S and an A100
# 80GB on 19 September 2026. It is a narrow margin, so if a future job reports
# "CUDA driver version is insufficient" on some other node type, the driver moved
# or that node is older, and the fix is CUDA 12:
#   torch==2.13.0+cu129  torchaudio==2.11.0+cu129
#   --index-url https://download.pytorch.org/whl/cu129
# CUDA 12.x wheels run on any driver >= 525, so that is the safe fallback.
#
# NOTE: these pins are M3's, not a laptop's. An RTX 4070 on driver 610.62 takes
# either; see Setup in CLAUDE.md.

--index-url https://download.pytorch.org/whl/cu130
--extra-index-url https://pypi.org/simple

torch==2.13.0+cu130
torchaudio==2.11.0+cu130

# soundfile, not torchaudio.load. Doubly important here: torchaudio 2.11
# delegates that call to TorchCodec, which needs system FFmpeg libraries, and
# on a cluster there is no sudo to install them. soundfile bundles libsndfile.
soundfile==0.14.0

numpy==2.5.1
pandas==3.0.5
tqdm==4.70.0

opacus==1.6.0

# Flask is here only because verify_setup.py imports app.py to assert that the
# tensor the server builds is byte-identical to the training one. That parity
# check is worth running before a long job, and Flask is ~1MB. The server
# itself does not run on M3; `requests` (test_api.py only) is left out.
Flask==3.1.3
